// Start Face Recognition (Browser Camera Only) async function startFaceRecognition() { const cameraId = parseInt(cameraSelect.value) || 0; if (!userIdInput.value.trim()) { updateStatus('Please enter your User ID first', 'error'); return; } try { updateStatus('Starting camera... Please allow camera access.', 'info'); // Get user media stream from browser ONLY - no backend calls const constraints = { video: { width: { ideal: 640 }, height: { ideal: 480 }, facingMode: 'user' } }; const stream = await navigator.mediaDevices.getUserMedia(constraints); currentStream = stream; // Display video stream videoElement.srcObject = stream; videoElement.classList.remove('hidden'); cameraPlaceholder.classList.add('hidden'); // Update UI startCameraBtn.disabled = true; stopCameraBtn.disabled = false; captureFaceBtn.disabled = false; cameraSelect.disabled = true; updateStatus('✅ Camera started! Position your face in the center of the frame.', 'success'); } catch (error) { console.error('Error starting camera:', error); if (error.name === 'NotAllowedError') { updateStatus('❌ Camera access denied. Please allow camera permissions and try again.', 'error'); } else if (error.name === 'NotFoundError') { updateStatus('❌ No camera found. Please check if your camera is connected.', 'error'); } else if (error.name === 'NotSupportedError') { updateStatus('❌ Camera not supported in this browser.', 'error'); } else { updateStatus(`❌ Camera error: ${error.message}`, 'error'); } } } // Stop camera function stopCamera() { if (currentStream) { currentStream.getTracks().forEach(track => track.stop()); currentStream = null; } // Update UI videoElement.srcObject = null; videoElement.classList.add('hidden'); cameraPlaceholder.classList.remove('hidden'); startCameraBtn.disabled = false; stopCameraBtn.disabled = true; captureFaceBtn.disabled = true; cameraSelect.disabled = false; clearStatus(); } // Capture face and verify - MODIFIED to use canvas capture async function captureAndVerify() { if (!currentStream || !userIdInput.value.trim()) { updateStatus('Please start camera and enter User ID', 'error'); return; } try { updateStatus('📸 Capturing face...', 'info'); captureFaceBtn.disabled = true; // Small delay to ensure video frame is ready await new Promise(resolve => setTimeout(resolve, 100)); // Capture frame from browser video element const canvas = document.createElement('canvas'); const context = canvas.getContext('2d'); canvas.width = videoElement.videoWidth; canvas.height = videoElement.videoHeight; context.drawImage(videoElement, 0, 0, canvas.width, canvas.height); // Small delay to ensure image processing is complete await new Promise(resolve => setTimeout(resolve, 50)); // Convert to base64 const imageData = canvas.toDataURL('image/jpeg', 0.8); updateStatus('🔍 Verifying face...', 'info'); // Verify face using backend with the captured image const verifyResponse = await fetch('/api/face-verification/verify_face/', { method: 'POST', headers: { 'Content-Type': 'application/json', 'X-CSRFToken': getCsrfToken() }, body: JSON.stringify({ user_id: userIdInput.value, image: imageData }) }); const verifyData = await verifyResponse.json(); if (verifyData.success) { if (verifyData.verified) { updateStatus(`✅ Verification Successful! Confidence: ${verifyData.confidence.toFixed(2)}%`, 'success'); } else { updateStatus(`❌ Verification Failed: ${verifyData.message}`, 'error'); } } else { updateStatus(`❌ Error: ${verifyData.error}`, 'error'); } } catch (error) { console.error('Error during verification:', error); updateStatus(`❌ Verification error: ${error.message}`, 'error'); } finally { captureFaceBtn.disabled = false; } }
